import { useEffect, useState } from "react";

function App() {
    const [users, setUsers] = useState([]);

    useEffect(() => {
        fetch("https://jsonplaceholder.typicode.com/users")
            .then(res => res.json())
            .then(data => setUsers(data));
    }, []);

    return (
        <div>
            <h1>User List</h1>

            {users.map(u => (
                <p key={u.id}>
                    {u.name} - {u.email}
                </p>
            ))}
        </div>
    );
}

export default App;
